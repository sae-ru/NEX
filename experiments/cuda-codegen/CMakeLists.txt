cmake_minimum_required(VERSION 3.25)
project(NEXCudaEmissionExperiment LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(NEX_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Matching NEX source")
set(NEX_BUILD_ROOT "${NEX_SOURCE_ROOT}/build" CACHE PATH "Already built NEX frontend")
find_package(LLVM 22.1.0 REQUIRED CONFIG)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
find_library(NEX_FRONTEND_LIBRARY NAMES nex_frontend PATHS "${NEX_BUILD_ROOT}" NO_DEFAULT_PATH REQUIRED)
file(SHA256 "${NEX_FRONTEND_LIBRARY}" NEX_FRONTEND_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${NEX_FRONTEND_LIBRARY}")
add_executable(nex-cuda-emit main.cpp emitter.cpp)
target_include_directories(nex-cuda-emit PRIVATE "${NEX_SOURCE_ROOT}/include" "${NEX_SOURCE_ROOT}/runtime")
target_include_directories(nex-cuda-emit SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
target_compile_definitions(nex-cuda-emit PRIVATE NEX_EXPERIMENT_FRONTEND_SHA256="${NEX_FRONTEND_SHA256}")
target_link_libraries(nex-cuda-emit PRIVATE "${NEX_FRONTEND_LIBRARY}" LLVM SQLite::SQLite3 OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(nex-cuda-emit PRIVATE ntdll bcrypt advapi32)
endif()
