cmake_minimum_required(VERSION 3.25)
project(NEXTypedResourceHostBoundary LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
enable_testing()
add_subdirectory(../external-resources foundation)
include(probes.cmake)

get_filename_component(NEX_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(NEX_RUNTIME "${NEX_SOURCE}/runtime")
set(FFI_COMMON "${NEX_RUNTIME}/nex_ffi.cpp" "${NEX_RUNTIME}/nex_ffi_types.cpp"
  "${NEX_RUNTIME}/nex_ffi_resources.cpp" "${NEX_RUNTIME}/nex_external_resource.cpp" "${NEX_RUNTIME}/nex_resource_executor.cpp"
  "${NEX_RUNTIME}/nex_files.cpp" "${NEX_RUNTIME}/nex_json.cpp")
add_executable(existing_typed_ffi "${NEX_SOURCE}/tests/ffi_typed.cpp" "${NEX_SOURCE}/examples/ffi/producer.c" ${FFI_COMMON})
target_include_directories(existing_typed_ffi PRIVATE "${NEX_RUNTIME}")
target_link_libraries(existing_typed_ffi PRIVATE Threads::Threads)
add_test(NAME existing_nominal_C_ABI COMMAND existing_typed_ffi)
foreach(target existing_module existing_missing)
  add_library(${target} MODULE "${NEX_SOURCE}/tests/ffi_module.c")
  target_include_directories(${target} PRIVATE "${NEX_RUNTIME}")
  set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_VISIBILITY_PRESET hidden)
endforeach()
target_compile_definitions(existing_missing PRIVATE NEX_FFI_TEST_NO_ENTRY)
add_executable(existing_loader "${NEX_SOURCE}/tests/ffi_loader.cpp" "${NEX_RUNTIME}/nex_ffi_loader.cpp" ${FFI_COMMON})
target_include_directories(existing_loader PRIVATE "${NEX_RUNTIME}")
target_link_libraries(existing_loader PRIVATE Threads::Threads OpenSSL::Crypto ${CMAKE_DL_LIBS})
add_test(NAME existing_pinned_module_loader COMMAND existing_loader $<TARGET_FILE:existing_module> $<TARGET_FILE:existing_missing>)
add_library(existing_foreign_module MODULE "${NEX_SOURCE}/tests/foreign_module.c")
target_include_directories(existing_foreign_module PRIVATE "${NEX_RUNTIME}")
set_target_properties(existing_foreign_module PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_VISIBILITY_PRESET hidden)
add_executable(existing_foreign_scope "${NEX_SOURCE}/tests/foreign_scope.cpp" "${NEX_RUNTIME}/nex_foreign.cpp"
  "${NEX_RUNTIME}/foreign_runtime.cpp" "${NEX_RUNTIME}/runtime.cpp" "${NEX_RUNTIME}/nex_ffi_loader.cpp" ${FFI_COMMON})
target_include_directories(existing_foreign_scope PRIVATE "${NEX_RUNTIME}")
target_link_libraries(existing_foreign_scope PRIVATE Threads::Threads OpenSSL::Crypto ${CMAKE_DL_LIBS})
add_test(NAME existing_foreign_lexical_scope COMMAND existing_foreign_scope $<TARGET_FILE:existing_foreign_module>)
