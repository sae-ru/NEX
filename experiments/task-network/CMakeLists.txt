cmake_minimum_required(VERSION 3.25)
project(NEXTaskNetwork LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
get_filename_component(NEX_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(task_network "${NEX_SOURCE}/tests/task_network.cpp"
  "${NEX_SOURCE}/runtime/nex_task_network.cpp"
  "${NEX_SOURCE}/runtime/nex_task_external.cpp")
target_include_directories(task_network PRIVATE "${NEX_SOURCE}/runtime")
target_compile_options(task_network PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(task_network PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(task_network PRIVATE ws2_32)
endif()
enable_testing()
add_test(NAME bounded_task_TCP_transport COMMAND task_network)
set_tests_properties(bounded_task_TCP_transport PROPERTIES TIMEOUT 90)
