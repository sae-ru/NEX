cmake_minimum_required(VERSION 3.25)
project(NEXWindowsRuntime LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_HOST_WIN32)
  message(FATAL_ERROR "This suite must execute on native Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NEX_WINDOWS_ASAN "Instrument the direct Windows runtime suite with Clang ASan" OFF)
option(NEX_WINDOWS_JOB_OBSERVER_ONLY "Build only the native job observer and its ownership calibration" OFF)
if(NEX_WINDOWS_ASAN)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "NEX_WINDOWS_ASAN requires the selected Clang sanitizer runtime")
  endif()
  add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address)
endif()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_executable(nex_windows_job_observer job_observer.cpp)
target_link_libraries(nex_windows_job_observer PRIVATE advapi32)
if(MINGW)
  target_link_options(nex_windows_job_observer PRIVATE -municode)
  if(NOT NEX_WINDOWS_ASAN)
    target_link_options(nex_windows_job_observer PRIVATE -static)
  endif()
endif()
enable_testing()
set(NEX_WINDOWS_PACKAGED_PYTHON "" CACHE FILEPATH "Optional Store Python for suspended packaged-root refusal calibration")
set(NEX_WINDOWS_PACKAGED_ARGS)
if(NEX_WINDOWS_PACKAGED_PYTHON)
  list(APPEND NEX_WINDOWS_PACKAGED_ARGS --packaged-python "${NEX_WINDOWS_PACKAGED_PYTHON}")
endif()
add_test(NAME windows_job_object_ownership COMMAND ${Python3_EXECUTABLE} -B
  ${CMAKE_CURRENT_LIST_DIR}/job_observer_test.py --observer $<TARGET_FILE:nex_windows_job_observer>
  ${NEX_WINDOWS_PACKAGED_ARGS})
set_tests_properties(windows_job_object_ownership PROPERTIES TIMEOUT 90)
set(NEX_WINDOWS_APPCONTAINER_PYTHON_ROOT "" CACHE PATH "Explicit compact ordinary Python runtime for real AppContainer calibration (standard-user broker required)")
set(NEX_WINDOWS_APPCONTAINER_LEGACY_OBSERVER "" CACHE FILEPATH "Optional calibrated one-hour observer for AppContainer compatibility checks")
if(NEX_WINDOWS_APPCONTAINER_PYTHON_ROOT)
  set(NEX_WINDOWS_APPCONTAINER_LEGACY_ARGS)
  if(NEX_WINDOWS_APPCONTAINER_LEGACY_OBSERVER)
    list(APPEND NEX_WINDOWS_APPCONTAINER_LEGACY_ARGS --legacy-observer "${NEX_WINDOWS_APPCONTAINER_LEGACY_OBSERVER}")
  endif()
  add_test(NAME windows_appcontainer_ownership COMMAND ${Python3_EXECUTABLE} -B
    ${CMAKE_CURRENT_LIST_DIR}/appcontainer_test.py --observer $<TARGET_FILE:nex_windows_job_observer>
    --runtime ${NEX_WINDOWS_APPCONTAINER_PYTHON_ROOT} ${NEX_WINDOWS_APPCONTAINER_LEGACY_ARGS})
  set_tests_properties(windows_appcontainer_ownership PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
endif()
# This fixture deliberately bypasses C++ destructors. Its standalone normal
# runtime is separate from the optional ASan IO suite.
if(NEX_WINDOWS_APPCONTAINER_PYTHON_ROOT AND NOT NEX_WINDOWS_ASAN)
  add_executable(nex_windows_appcontainer_abrupt appcontainer_abrupt.cpp)
  target_include_directories(nex_windows_appcontainer_abrupt PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../runtime")
  target_link_libraries(nex_windows_appcontainer_abrupt PRIVATE advapi32 bcrypt)
  if(MINGW)
    target_link_options(nex_windows_appcontainer_abrupt PRIVATE -static -municode)
  endif()
  add_test(NAME windows_appcontainer_owned_cleanup COMMAND ${Python3_EXECUTABLE} -B
    ${CMAKE_CURRENT_LIST_DIR}/appcontainer_cleanup_test.py
    --observer $<TARGET_FILE:nex_windows_job_observer>
    --abrupt-probe $<TARGET_FILE:nex_windows_appcontainer_abrupt>)
  set_tests_properties(windows_appcontainer_owned_cleanup PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
endif()
set(NEX_WINDOWS_APPCONTAINER_CODE_PARENT "" CACHE PATH "Explicit existing staging parent, traversable/enumerable by AppContainer; existing ACLs are never changed")
if(NEX_WINDOWS_APPCONTAINER_PYTHON_ROOT AND NEX_WINDOWS_APPCONTAINER_CODE_PARENT AND NOT NEX_WINDOWS_ASAN)
  add_executable(nex_windows_appcontainer_path appcontainer_path.cpp)
  if(MINGW)
    target_link_options(nex_windows_appcontainer_path PRIVATE -static -municode)
  endif()
  add_test(NAME windows_appcontainer_private_code_root COMMAND ${Python3_EXECUTABLE} -B
    ${CMAKE_CURRENT_LIST_DIR}/code_root_test.py
    --observer $<TARGET_FILE:nex_windows_job_observer>
    --probe $<TARGET_FILE:nex_windows_appcontainer_path>
    --runtime "${NEX_WINDOWS_APPCONTAINER_PYTHON_ROOT}"
    --code-parent "${NEX_WINDOWS_APPCONTAINER_CODE_PARENT}")
  set_tests_properties(windows_appcontainer_private_code_root PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
endif()
if(NEX_WINDOWS_JOB_OBSERVER_ONLY)
  return()
endif()
set(NEX_SQLITE_SOURCE "" CACHE FILEPATH "Pinned SQLite amalgamation sqlite3.c")
if(NOT EXISTS "${NEX_SQLITE_SOURCE}")
  message(FATAL_ERROR "Set NEX_SQLITE_SOURCE to the pinned SQLite amalgamation")
endif()
get_filename_component(NEX_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(NEX_SQLITE_INCLUDE "${NEX_SQLITE_SOURCE}" DIRECTORY)
add_library(nex_windows_sqlite STATIC "${NEX_SQLITE_SOURCE}")
target_compile_definitions(nex_windows_sqlite PRIVATE SQLITE_THREADSAFE=1)
add_executable(nex_windows_io "${NEX_ROOT}/tests/windows_io.cpp"
  "${NEX_ROOT}/runtime/nex_files.cpp" "${NEX_ROOT}/runtime/nex_store.cpp"
  "${NEX_ROOT}/runtime/nex_resource_region.cpp"
  "${NEX_ROOT}/runtime/nex_external_resource.cpp"
  "${NEX_ROOT}/runtime/nex_resource_executor.cpp")
target_include_directories(nex_windows_io PRIVATE "${NEX_ROOT}/runtime" "${NEX_SQLITE_INCLUDE}")
target_link_libraries(nex_windows_io PRIVATE nex_windows_sqlite ntdll bcrypt advapi32)
if(MINGW AND NOT NEX_WINDOWS_ASAN)
  target_link_options(nex_windows_io PRIVATE -static)
endif()
enable_testing()
add_test(NAME windows_ntfs_runtime COMMAND nex_windows_io)
set_tests_properties(windows_ntfs_runtime PROPERTIES TIMEOUT 60)
set(NEX_CUDA_INCLUDE "" CACHE PATH "Optional installed CUDA Toolkit include directory")
if(NEX_CUDA_INCLUDE)
  if(NOT EXISTS "${NEX_CUDA_INCLUDE}/cuda.h")
    message(FATAL_ERROR "NEX_CUDA_INCLUDE must contain the installed cuda.h")
  endif()
  add_executable(nex_windows_cuda_probe cuda_probe.cpp)
  target_include_directories(nex_windows_cuda_probe PRIVATE "${NEX_CUDA_INCLUDE}" "${NEX_ROOT}/runtime")
  if(MINGW AND NOT NEX_WINDOWS_ASAN)
    target_link_options(nex_windows_cuda_probe PRIVATE -static)
  endif()
  add_test(NAME windows_cuda_prerequisite COMMAND nex_windows_cuda_probe)
  set_tests_properties(windows_cuda_prerequisite PROPERTIES TIMEOUT 120)
endif()
